cmake_minimum_required(VERSION 3.14)
project(MedicineDistributionPrioritySystem CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)

# backend/main.cpp includes saksham.cpp, which includes every member .cpp (single translation unit).
add_executable(medicine_server backend/main.cpp)
target_include_directories(medicine_server PRIVATE ${SQLite3_INCLUDE_DIRS})
target_link_libraries(medicine_server PRIVATE ${SQLite3_LIBRARIES} Threads::Threads)
if(WIN32)
  target_link_libraries(medicine_server PRIVATE ws2_32)
endif()
if(NOT MSVC)
  target_compile_options(medicine_server PRIVATE -Wall -Wextra)
endif()

add_executable(unit_tests tests/unit_tests.cpp)
target_link_libraries(unit_tests PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(unit_tests PRIVATE ws2_32)
endif()
enable_testing()
add_test(NAME unit_tests COMMAND unit_tests)
